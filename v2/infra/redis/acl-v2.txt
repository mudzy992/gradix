# Gradix v2 staging — read-only audit ugovora Redis ACL-a.
# Ovu provjeru izvršava vlasnik infrastrukture privilegovanim redis-cli nalogom;
# agent ne pristupa Redis serveru niti mijenja ACL.
#
# LIVE korisnik `gradix_v2` već postoji i napravljen je prema ovom ugovoru.
# Ovaj fajl je namjerno samo za pregled: NE pokretati ACL SETUSER, posebno ne sa `reset`;
# to bi moglo resetovati postojeću lozinku, ključeve i dozvole. Ne kreirati niti rotirati
# live korisnika u okviru ove izmjene. Ako se live stanje razlikuje, zaustaviti se i prvo
# dogovoriti zasebnu, kontrolisanu ACL promjenu.
#
# Očekivani efektivni ugovor za `gradix_v2`:
#   status: on
#   key pattern: ~gradix:stg:*
#   commands: +ping +set +pttl +get +del
# Bez `~*`, `+@all`, pristupa v1 namespace-u, KEYS/SCAN ili administrativnih komandi.
# Trenutna dijagnostika koristi samo PING i prefiksirani SET/PTTL/GET/DEL.
# Redis TLS ostaje na 6380; CA i REDIS_TLS_CA_FILE putanje su opisane u docs/COOLIFY.md.

# Read-only provjera postojećeg korisnika (pokrenuti kao Redis ACL administrator):
ACL GETUSER gradix_v2

# Staging web kontejner koristi:
#   REDIS_HOST=redis-core (isti external redis_net)
#   REDIS_PORT=6380
#   REDIS_USER=gradix_v2
#   REDIS_PREFIX=gradix:stg:
#   REDIS_TLS_CA_FILE=/etc/ssl/certs/redis/redis-ca.crt
